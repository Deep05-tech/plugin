# AI Content Schema Specification (v1.0) — SEO City Pages

## 1. Overview

The content generated by the AI Content Engine follows a strict, versioned JSON Schema conforming to the OpenAI Structured Outputs API (`response_format: { type: "json_schema", json_schema: { strict: true } }`).

All fields are strictly typed, mandatory (`required`), and disallowed from receiving arbitrary additional properties (`additionalProperties: false`).

---

## 2. JSON Schema Structure

```json
{
  "name": "seo_city_landing_page_content",
  "strict": true,
  "schema": {
    "type": "object",
    "required": [
      "hero",
      "introduction",
      "service_overview",
      "benefits",
      "process",
      "local_relevance",
      "faqs",
      "final_cta"
    ],
    "additionalProperties": false,
    "properties": {
      "hero": {
        "type": "object",
        "required": ["title", "description", "cta_text"],
        "additionalProperties": false,
        "properties": {
          "title": { "type": "string" },
          "description": { "type": "string" },
          "cta_text": { "type": "string" }
        }
      },
      "introduction": {
        "type": "object",
        "required": ["heading", "body"],
        "additionalProperties": false,
        "properties": {
          "heading": { "type": "string" },
          "body": { "type": "string" }
        }
      },
      "service_overview": {
        "type": "object",
        "required": ["heading", "body"],
        "additionalProperties": false,
        "properties": {
          "heading": { "type": "string" },
          "body": { "type": "string" }
        }
      },
      "benefits": {
        "type": "array",
        "items": {
          "type": "object",
          "required": ["title", "description"],
          "additionalProperties": false,
          "properties": {
            "title": { "type": "string" },
            "description": { "type": "string" }
          }
        }
      },
      "process": {
        "type": "array",
        "items": {
          "type": "object",
          "required": ["step_number", "title", "description"],
          "additionalProperties": false,
          "properties": {
            "step_number": { "type": "integer" },
            "title": { "type": "string" },
            "description": { "type": "string" }
          }
        }
      },
      "local_relevance": {
        "type": "object",
        "required": ["heading", "body"],
        "additionalProperties": false,
        "properties": {
          "heading": { "type": "string" },
          "body": { "type": "string" }
        }
      },
      "faqs": {
        "type": "array",
        "items": {
          "type": "object",
          "required": ["question", "answer"],
          "additionalProperties": false,
          "properties": {
            "question": { "type": "string" },
            "answer": { "type": "string" }
          }
        }
      },
      "final_cta": {
        "type": "object",
        "required": ["heading", "description", "button_text"],
        "additionalProperties": false,
        "properties": {
          "heading": { "type": "string" },
          "description": { "type": "string" },
          "button_text": { "type": "string" }
        }
      }
    }
  }
}
```
